```from time import *

def v2(n):
    result = 0
    while n%2 == 0:
        n = n >> 1
        result += 1
    return result

def three_exp(n, bits):
    if n == 0:
        return 1
    if n == 1:
        return 3

    if n % 2 == 0:
        return (three_exp(n // 2, bits) ** 2) % (2 ** bits)
    else:
        return (3 * three_exp(n // 2, bits) ** 2) % (2 ** bits)

def k(i, bits):
    return (2 * three_exp(i, bits) + i + 5) % (2 ** bits)

def v2k(i, start_bits):
    while k(i, start_bits) == 0:
        start_bits += 1
    return start_bits - 1

first_seen = []
for i in range(1000):
    first_seen.append(-1)

i = 1

while current_exponent < 900:
    val = v2k(i, current_exponent)
    if not val in seen_exponents:
        seen_exponents.add(val)
        first_seen[val] = i
    if val == current_exponent:
        i += 2 ** current_exponent
        while current_exponent in seen_exponents:

    else:
        i += 2 ** current_exponent
    print(i)
```
